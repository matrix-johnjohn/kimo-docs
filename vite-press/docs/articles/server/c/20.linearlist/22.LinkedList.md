## 链表的概念
在链表中,结点在存储器中的位置是任意的,即逻辑上相邻的数据元素在物理上不一定相邻;
![](https://raw.githubusercontent.com/matrix-johnjohn/kimo-docs/refs/heads/main/images/articles/server/c/data-struct/linearlist/3.linearlist.png)
<br>
指针,头节点,和首元结点;<br>
`头指针`是指向链表中的第一个结点的指针;`头节点`是在链表的首元结点之前附设的一个结点,数据域只放空表标志和表长等信息;`首元结点`是指链表中存储第一个数据元素的结点;
![](https://raw.githubusercontent.com/matrix-johnjohn/kimo-docs/refs/heads/main/images/articles/server/c/data-struct/linkedlist/02.linkedlist.png)

## 链表的结点
链表的结点是数据元素的存储映像,由`数据域`和`指针域`两部分组成;
![](https://raw.githubusercontent.com/matrix-johnjohn/kimo-docs/refs/heads/main/images/articles/server/c/data-struct/linkedlist/01.linkedlist.png)

## 链表的结构
定义一个链表,`LinkList L`为链表的头指针,L对应的结点变量表示链表上的每一个结点;
````C
typedef struct LNode {
	ElemType data;
	struct LNode* next;
} LNode, * LinkList;
````

## 链表的初始化
初始化一个空表,带头结点的单链表;<br>
生成新节点作为头节点,用头指针`list`指向`头节点`,`头节点`的`指针域置空`;
````C
void InitLinkedList(LinkedList* L){//初始化链表
    *L = (LinkedList)malloc(sizeof(LNode));
    if (!*L) {
        printf("Memory allocation failed!\n");
        exit(EXIT_FAILURE);
    }
    (*L)->next = NULL;
};
````

## 判断链表是否非空
传入链表的头指针,根据头指针指向的头结点的下一个结点是否为`NULL`;
````C
int isEmpty(LinkedList *L) {//判断链表是否为空
	LinkedList p = (*L);
    if (p->next == NULL) {
        return 1;
    }
    else {
        return 0;
	}
};
````

## 链表的销毁
````C
void DestroyList(LinkedList* L){//销毁链表
    LNode* p = (*L);
    while((*L)!=NULL){
		p = *L;
        *L = (*L)->next;
		free(p);
    }
};
````

## 链表的长度
从首元结点开始,一次计数所有结点;
````C
int GetLength(LinkedList L){//获取链表长度
    if (L == NULL)return -1;
    LNode* p = L;
    int length = 0;
    while(p!=NULL){
		length++;
		p=p->next;
    }
	return length;
};
````

````C
LNode* LocateElem(LinkList list,ElemType elem){
````

## 链表的插入
````C
void InsertLinkList(LinkList list,ElemType elem,int index){
	
	int currIdx=0;
	LNode* p=list;
	
	while(p!=NULL&&currIdx<index){
		p = p->next;
		currIdx++;
	}

	LNode *s=(LNode*)malloc(sizeof(LNode));
	s->value=elem;
	s->next=p->next;
	p->next = s;
}
````

## 链表的删除

````C
void DeleteElem(LinkedList *L,int index){//删除指定位置元素
    int i=0;
	LinkedList p=*L;
    while (p != NULL&&i<index){
		p = p->next;
        i++;
    };

};
````

## 链表的尾插法
````C
void CreateList_R(LinkedList* L,int len){
	*L = (LNode*)malloc(sizeof(LNode));//创建头结点
    (*L)->next=NULL;
	LNode* R = *L;//尾指针初始化,指向头结点
    for(int i=0;i<len;i++){
		LNode* p = (LNode*)malloc(sizeof(LNode));//创建新节点
        p->data = i+1;//赋值
		p->next = R->next;
		R->next = p;//将新节点插入到尾部
		R = p;//更新尾指针
    };
};
````

## 链表的头插法
````C
void CreateList_H(LinkedList* L,int length){//头插法创建链表
    *L=(LinkedList)malloc(sizeof(LNode));
    (*L)->next=NULL;
    LinkedList curr=(*L);
    for(int i=0;i<length;i++){
        LNode *p=(LNode*)malloc(sizeof(LNode));
        p->data = i;
        p->next=curr->next;
        curr->next=p;
    };
};
````

## 链表完整实现
````C
#define _CRT_SECURE_NO_WARNINGS
#include<stdio.h>
#include<stdlib.h>



typedef struct LNode {
    int data;
    struct LNode* next;
}LNode, * LinkList;

void InitLinkList(LinkList* L, LinkList* R) {//初始化链表
    *L = (LinkList)malloc(sizeof(LNode));
    (*L)->next = NULL;
    *R = (*L);
};

void DestroyList(LinkList* L) {//销毁链表
    LNode* p = (*L);
    while ((*L) != NULL) {
        p = *L;
        *L = (*L)->next;
        free(p);
    };
};

void InsertList_R(LinkList* L, LinkList* R, int data) {//链表结点尾插
    LNode* s = (LNode*)malloc(sizeof(LNode));
    s->data = data;
    //2.链表链接
    s->next = (*R)->next;
    (*R)->next = s;
    (*R) = s;
};

void InsertList_H(LinkList* L, int data) {//链表结点头插
    //1.初始化数据
    LNode* s = (LNode*)malloc(sizeof(LNode));
    s->data = data;
    //2.链表链接
    s->next = (*L)->next;
    (*L)->next = s;
};

LNode* GetElem(LinkList* L, int i) {//查找结点
    int j = 0;
    LNode* p = (*L)->next;
    while (p != NULL && j < i) {
        j++;
        p = p->next;
    }
    return p;
};

void ListDelete(LinkList* L, int i) {//删除结点
    int j = 0;
    LNode* p = (*L);
    while (p != NULL && j < i) {
        j++;
        p = p->next;
    }
    LNode* s = p->next;
    p->next = p->next->next;
    free(s);
}

void Traversal(LinkList L) {//遍历
    LNode* p = L->next;
    while (p != NULL) {
        printf("%d\n", p->data);
        p = p->next;
    }
};

void Traversal_R(LinkList L){
    if (L == NULL)return;
    printf("%d ",L->data);
    Traversal_R(L->next);
};

int main() {
    LinkList L=NULL, R=NULL;
    InitLinkList(&L,&R);
    for (int i = 1; i <= 10; i++) {
        InsertList_R(&L,&R,i);
    };
    Traversal_R(L->next);
	return 0;
};
````
