## 双向链表
单链表的链式存储结构的结点中只有一个指向直接后继的指针域,由此,从某个结点触发只能顺着指针向后寻找其他结点;

## 双向链表的存储结构
````C
typedef struct DulNode{
    int data;
	struct DulNode* prior;
    struct DulNode* next;
}DulNode,*DulLinkList;
````

## 双向链表的插入
````C
void ListInsert_DulLinkList(DulLinkList *L,int i,int e){//插入双向链表
    if ((*L)->next == NULL) {
        exit(EXIT_FAILURE);
    }
	DulNode* p=(*L)->next;
    int j = 0;
    while (p!=NULL&&j<i){
        p=p->next;
        j++;
    }
    DulNode *s=(DulNode*)malloc(sizeof(DulNode));
	s->data = e;
    s->prior = p->prior;
    p->prior->next = s;
    s->next = p;
    p->prior = s;
}
````

## 双向链表的删除
````C
void ListDelete_DulLinkList(DulLinkList* L, int i) {//删除双向链表
    DulNode* p = (*L)->next;
    int j = 0;
    while (p != NULL && j < i) {
        p = p->next;
        j++;
    };
    p->prior->next = p->next;
	p->next->prior = p->prior;
    free(p);
}
````

## 双向链表的完整实现
````C
#include "common.h"

typedef struct DuLNode{
    int data;
    struct DuLNode* prior;
    struct DuLNode* next;
}DuLNode,*DuLinkList;

Status InitDuLinkList(DuLinkList* L, DuLinkList* R) {//初始化双向链表
    (*L)=(DuLinkList)malloc(sizeof(DuLNode));
    if(!(*L))return ERROR;
    (*L)->next=NULL;
	(*L)->prior=NULL;
    (*R)=(*L);
	return SUCCESS;
};

void DestroyDuLinkList(DuLinkList *L){//销毁双向链表
    DuLNode *p=(*L);
    while ((*L)!=NULL){
        p = (*L);
		*L = (*L)->next;
        free(p);
    }
};

void InsertList_R(DuLinkList* L, DuLinkList* R, int data) {//在双向链表尾部插入元素
	//1.创建新节点
    DuLNode *s=(DuLNode*)malloc(sizeof(DuLNode));
    s->data=data;
    s->next = (*R)->next;
	//2.链接链表
    (*R)->next = s;
    s->prior = (*R);
    (*R) = s;
};

void InsertDuLinkList_H(DuLinkList* L,DuLinkList *R,int data) {//在双向链表头部插入元素
	//1.创建新节点
    DuLNode *s=(DuLNode*)malloc(sizeof(DuLNode));
    s->data=data;
    //2.链接链表
	s->next = (*L)->next;
    if ((*L)->next!=NULL){
        (*L)->next->prior = s;
    }
    if ((*L)->next==NULL){
        *R = s;
    }
	(*L)->next = s;
	s->prior = (*L);
};

void InsertDuLinkList(DuLinkList *L,int i,int data){//插入链表
	DuLNode* p = (*L)->next;
    int j = 0;
    while (p!=NULL && j<i){
        p = p->next;
        j++;
	};
    DuLNode *s=(DuLNode*)malloc(sizeof(DuLNode));
    s->data = data;
    s->prior = p->prior;
    s->prior->next = s;
	s->next = p;
    p->prior = s;
};

void DeleteDuLinkList(DuLinkList *L,int i){//删除链表
    DuLNode* p = (*L)->next;
    int j = 0;
    while (p!=NULL && j<i){
        p = p->next;
        j++;
    };
    p->prior->next = p->next;
	if (p->next != NULL)
	    p->next->prior = p->prior;
	free(p);
};  

void Traversal(DuLinkList* L) {//遍历双向链表
    DuLNode *p=(*L)->next;
    while(p!=NULL){
        printf("%d ",p->data);
		p = p->next;
    };
};

int main() {
	DuLinkList L=NULL,R=NULL;
	InitDuLinkList(&L,&R);
    for (int i = 0; i <3; i++) {
        InsertDuLinkList_H(&L,&R, i + 1);
    }
	DeleteDuLinkList(&L, 2);
	Traversal(&L);
	DestroyDuLinkList(&L);
    return 0;
}
````
