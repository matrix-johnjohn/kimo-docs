int Partition(int a[],int low,int height){
    int pivote=a[low];
    while(low<height){
        while(low<height&&a[height]>pivote)height--;
        a[low]=a[height];
        while(low<height&&a[low]<pivote)low++;
        a[height]=a[low];
    }
    a[low]=pivote;
    return low;
};

void quick_sort(int a[],int low,int height){
    if(low<height){
        int pivote=Partition(a,low,height);
        quick_sort(a,low,pivote-1);
        quick_sort(a,pivote+1,height);    
    }
}
