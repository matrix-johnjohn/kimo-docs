## 分治函数
````c
void merge(int a[],int t[],int low,int mid,int height){
    int i=low,j=mid+1,k=low;
    while(i<=mid&&j<=height){
        if(a[i]>a[j]){
            t[k++]=a[j++];
        }else{
            t[k++]=a[i++];
        }
    }
    while(i<=mid){
        t[k++]=a[i++];
    };
    while(j<=height){
        t[k++]=a[j++];
    };
    for(int i=low;i<=height;i++){
        a[i]=t[i];
    }
}
````

## 归并函数
````c
void merge_sort(int a[],int t[],int low,int height){
    if(low>=height)return;
    int mid=(low+height)/2;
    merge_sort(a,t,low,mid);
    merge_sort(a,t,mid+1,height);
    merge(a,t,low,mid,height);
};
````
