`栈`是限定仅在`栈顶插入`或`删除`操作的线性表;栈的插入操作被称为`进栈`(push),栈的删除操作被称为`出栈`(pop);
![](https://raw.githubusercontent.com/matrix-johnjohn/kimo-docs/refs/heads/main/images/articles/server/c/data-struct/stack/01.stack.png)


## 链栈
链栈是指采用`链式存储结构`实现的栈,通常用单指针单链表作为其存储结构,用头指针对该存储结构进行`添加`或`删除`元素;
## 链栈存储结构
````C
typedef struct StackNode{
    int data;
    struct StackNode *next;
}StackNode,*LinkStack;
````

## 链栈初始化
````C
Status InitLinkStack(LinkStack *S){
    (*S)=NULL;
    return OK;
};
````

## 链栈入栈
````C
Status Push(LinkStack *S,int data){
    StackNode *p=(StackNode*)malloc(sizeof(StackNode));
    if(p==NULL)return ERROR;
    p->data=data;
    p->next=(*S);
    (*S)=p;
    return OK;
};
````

## 链栈出栈
````C
Status Pop(LinkStack *S,int *data){
    if(*S==NULL)return OVERFLOW;
    StackNode *p=*S;
    *S=p->next;
    *data=p->data;
    free(p);
    return OK;
};
````

## 获取链栈头元素
````C
int GetTop(LinkStack S){
    if(S==NULL)return OVERFLOW;
    return S->data;
}
````

## 递归遍历链栈
````C
void Traversal_R(LinkStack S){
    if(S==NULL){
        printf("\n");
        return;
    };
    printf("%d ",S->data);
    Traversal_R(S->next);
};
````

## 链栈完整代码实现
````C
#include<stdio.h>
#include<stdlib.h>
#include "common.h"

typedef struct StackNode{
    int data;
    struct StackNode *next;
}StackNode,*LinkStack;

Status InitLinkStack(LinkStack *S){
    (*S)=NULL;
    return OK;
};

Status Push(LinkStack *S,int data){
    StackNode *p=(StackNode*)malloc(sizeof(StackNode));
    if(p==NULL)return ERROR;
    p->data=data;
    p->next=(*S);
    (*S)=p;
    return OK;
};

Status Pop(LinkStack *S,int *data){
    if(*S==NULL)return OVERFLOW;
    StackNode *p=*S;
    *S=p->next;
    *data=p->data;
    free(p);
    return OK;
};

int GetTop(LinkStack S){
    if(S==NULL)return OVERFLOW;
    return S->data;
};

void Traversal_R(LinkStack S){
    if(S==NULL){
        printf("\n");
        return;
    };
    printf("%d ",S->data);
    Traversal_R(S->next);
};

int main(){
    LinkStack S=NULL;
    int *data=(int*)malloc(sizeof(int));
    InitLinkStack(&S);
    for(int i=1;i<=5;i++){
        Push(&S,i);
    };
    Pop(&S,data);
    Traversal_R(S);
    printf("%d %d\n",*data,GetTop(S));
    Traversal_R(S);
    return 0;
}
````
