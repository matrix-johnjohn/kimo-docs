## 存储结构
````c
typedef struct Node{
    int data;
    struct Node* next;
}Node;

typedef struct Queue{
    Node* front;
    Node* rear;
}Queue,*QueuePtr;
````

## 构建优先级队列
````c
void offer_queue(Queue* Q,int e){
    Node* p=Q->front;
    Node* s=(Node*)malloc(sizeof(Node));s->data=e;
    while(((p->next)!=NULL)&&(p->next->data>e)){
        p=p->next;
    }
    if(p==Q->rear){
        Q->rear->next=s;
        Q->rear=s;
    }else{
        s->next=p->next;
        p->next=s;
    }
    
};
````

## 优先级队列的删除操作
````c
int poll_queue(Queue* Q){
    Node* p=Q->front->next;
    int e=p->data;
    if(Q->front->next==Q->rear)Q->rear=Q->front;
    Q->front->next=Q->front->next->next;
    free(p);
    return e;
};
````
