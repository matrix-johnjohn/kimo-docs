队列的操作和栈类似,不同的是,队列的删除是在表的头部;


## 顺序队列
顺序队列`采取顺序存储结构`实现的队列称为`顺序队`,该结构除了用一组`地址连续的存储单元`依次存放从队头到队尾的元素之外,还有`front`和`rear`分别以索引的方式指向`队头`和`队尾`的位置;

## 顺序队列结构
````C
typedef struct SqQueue{
    int *base;
    int front;
    int rear;
}SqQueue;
````

## 顺序队列初始化
````C
Status InitSqQueue(SqQueue **Q){
    *Q=(SqQueue*)malloc(sizeof(SqQueue));
    if((*Q)==NULL)return OVERFLOW;
    (*Q)->base=(int*)malloc(sizeof(int)*MAXSIZE);
    (*Q)->rear=(*Q)->front=0;
    return OK;
};
````

## 顺序队列入队
````C
Status Push(SqQueue *Q,int data){
    if((Q->rear+1)%MAXSIZE==Q->front)return OVERFLOW;
    Q->base[Q->rear]=data;
    Q->rear=(Q->rear+1)%MAXSIZE;
    return OK;
};
````

## 顺序队列出队
````C
Status Pop(SqQueue *Q,int *data){
    if(Q->front==Q->rear)return OVERFLOW;
    *data=Q->base[Q->front];
    Q->front=(Q->front+1) %MAXSIZE;
    return OK;
};
````

## 获取队头
````C
int Peek(SqQueue *Q){
    if(Q->front==Q->rear)return OVERFLOW;
    return Q->base[Q->front];
}
````

## 顺序队列遍历
````C
void Traversal(SqQueue *Q){
    int front=Q->front;
    while(front!=Q->rear){
        printf("%d ",Q->base[front]);
        front=(front+1)%MAXSIZE;
    }
};
````

## 链式队列
采用`链式存储结构`的队列被称为`链队`,通常使用`单链表`表示;

## 队列的链式存储结构
````C
typedef struct QNode{//队列结点
    int data;
    struct QNode* next;
}QNode,*QueuePtr;

typedef struct LinkQueue{//队列结构
    QueuePtr front;
    QueuePtr rear;
}LinkQueue;
````

## 链队初始化
````C
Status InitLinkQueue(LinkQueue **Q){//初始化链队
    (*Q)=(LinkQueue*)malloc(sizeof(LinkQueue));
    if((*Q)==NULL)return OVERFLOW;
    (*Q)->front=(*Q)->rear=(QNode*)malloc(sizeof(QNode));
    (*Q)->front->next=NULL;
    return OK;
};
````

## 链队入队
````C
Status EnQueue(LinkQueue *Q,int data){//链队入队
    QNode *p=(QNode*)malloc(sizeof(QNode));
    if(p==NULL)return OVERFLOW;
    p->data=data;
    p->next=NULL;
    Q->rear->next=p;
    Q->rear=p;
    return OK;
};
````

## 链队出队
````C
Status DeQueue(LinkQueue *Q,int *data){//链队出队
    if(Q->front==Q->rear)return ERROR;
    QNode *p=Q->front->next;
    (*data)=p->data;
    Q->front->next=p->next;
    if(Q->rear==p)Q->rear=Q->front;
    free(p);
    return OK;
};
````

## 获取链队队头元素
````C
int GetHead(LinkQueue *Q){//取队头元素
    if(Q->front!=Q->rear)
    return Q->front->next->data;
    else 
    return OVERFLOW;
};
````

## 链队遍历
````C
void Traversal(LinkQueue *Q){//链队遍历
    QNode *p=Q->front->next;
    while(p!=NULL){
        printf("%d ",p->data);
        p=p->next;
    }
    printf("\n");
};
````

## 链队完整代码
````C
#include<stdio.h>
#include<stdlib.h>
#include "common.h"

typedef struct QNode{//队列结点
    int data;
    struct QNode* next;
}QNode,*QueuePtr;

typedef struct LinkQueue{//队列结构
    QueuePtr front;
    QueuePtr rear;
}LinkQueue;

Status InitLinkQueue(LinkQueue **Q){//初始化链队
    (*Q)=(LinkQueue*)malloc(sizeof(LinkQueue));
    if((*Q)==NULL)return OVERFLOW;
    (*Q)->front=(*Q)->rear=(QNode*)malloc(sizeof(QNode));
    (*Q)->front->next=NULL;
    return OK;
};

Status EnQueue(LinkQueue *Q,int data){//链队入队
    QNode *p=(QNode*)malloc(sizeof(QNode));
    if(p==NULL)return OVERFLOW;
    p->data=data;
    p->next=NULL;
    Q->rear->next=p;
    Q->rear=p;
    return OK;
};

Status DeQueue(LinkQueue *Q,int *data){//链队出队
    if(Q->front==Q->rear)return ERROR;
    QNode *p=Q->front->next;
    (*data)=p->data;
    Q->front->next=p->next;
    if(Q->rear==p)Q->rear=Q->front;
    free(p);
    return OK;
};

int GetHead(LinkQueue *Q){//取队头元素
    if(Q->front!=Q->rear)
    return Q->front->next->data;
    else 
    return OVERFLOW;
};

void Traversal(LinkQueue *Q){//链队遍历
    QNode *p=Q->front->next;
    while(p!=NULL){
        printf("%d ",p->data);
        p=p->next;
    }
    printf("\n");
};

int main(){
    int *data=(int*)malloc(sizeof(int));
    LinkQueue *Q=NULL;

    InitLinkQueue(&Q);

    for(int i=1;i<=10;i++){
        EnQueue(Q,i);
    };
    DeQueue(Q,data);
    Traversal(Q);
    printf("%d %d",*data,GetHead(Q));

    return 0;
};
````
