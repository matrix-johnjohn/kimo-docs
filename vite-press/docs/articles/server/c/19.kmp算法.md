## next数组
````cpp
vector<int> getNext(string p){
    if(p.size()<2){
        return {0};
    }
    vector<int>nexts(10,0);
    size_t comp=0,cur=2;
    while(cur<p.size()){
        if(p[cur-1]==p[comp]){
            nexts[cur]=comp+1;
            cur++;comp++;
        }else if(comp>0){
            comp=nexts[comp];
        }else if(comp==0){
            nexts[cur++]=0;
        };
    }
    return nexts;
};
````

## kmp匹配
````cpp
int kmp(string s,string p){
    vector<int> next=getNext(p);
    int i=0;
    int j=0;
    while(i<s.size()&&j<p.size()){
        if(s[i]==s[j]){
            i++;
            j++;
        }else if(j>0){
            j=next[j];
        }else{
            i++;
        }
    }
    return j==p.size()?i-p.size():-1;
}
````
