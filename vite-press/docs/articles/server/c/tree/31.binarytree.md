## 完整代码
````C
#include<stdio.h>
#include<stdlib.h>
#include "index.h"

typedef struct TreeNode{//结点结构
    int data;
    struct TreeNode *left;
    struct TreeNode *right;
}TreeNode,*BinaryTree;

Status InitBinaryTree(BinaryTree *root,int data){//创建根节点
    *root=(TreeNode*)malloc(sizeof(TreeNode));
    (*root)->data=data;
    if(root==NULL)return OVERFLOW;
    return OK;
};

TreeNode* LeftInsert(BinaryTree *root,int data){//左侧插入节点
    TreeNode *p=(TreeNode*)malloc(sizeof(TreeNode));
    p->data=data;
    (*root)->left=p;
    return p;
};

TreeNode* RightInsert(BinaryTree *root,int data){//右侧插入节点
    TreeNode *p=(TreeNode*)malloc(sizeof(TreeNode));
    p->data=data;
    (*root)->right=p;
    return p;
};

void preOrderTraversal(TreeNode *root){//先序遍历
    if(root==NULL)return;
    printf("%d ",root->data);
    preOrderTraversal(root->left);
    preOrderTraversal(root->right);
};

void inOrderTraversal(TreeNode *root){//中序遍历
    if(root==NULL)return;
    inOrderTraversal(root->left);
    printf("%d ",root->data);
    inOrderTraversal(root->right);
};

void postOrderTraversal(TreeNode *root){//后序遍历
    if(root==NULL)return;
    postOrderTraversal(root->left);
    postOrderTraversal(root->right);
    printf("%d ",root->data);
}

int max(int leftCount,int rightCount){
    return (leftCount>rightCount?leftCount:rightCount)+1;
}

int LayerCount(TreeNode *root){//二叉树的最大深度
    if(root==NULL)return 0;
    int leftCount=LayerCount(root->left);
    int rightCount=LayerCount(root->right);
    return max(leftCount,rightCount);
};

int LeadCount(TreeNode *root){//二叉树叶子节点数量
    if(root==NULL)return 0;
    if(root->left==NULL&&root->right==NULL)return 1;
    int left=LeadCount(root->left);
    int right=LeadCount(root->right);
    return left+right;
};

int NodeCount(TreeNode *root){//二叉树的结点数量
    if(root==NULL)return 0;
    int left=NodeCount(root->left);
    int right=NodeCount(root->right);
    return left+right+1;
};

int main(){
    BinaryTree root;
    InitBinaryTree(&root,1);
    TreeNode *p1=LeftInsert(&root,2);
    TreeNode *p2=RightInsert(&root,3);
    TreeNode *p3=LeftInsert(&p1,4);
    TreeNode *p4=RightInsert(&p1,5);
    return 0;
}
````
