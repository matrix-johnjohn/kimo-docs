哈夫曼(Huffman)树又称最优树,是一类带权路径长度最短的树,树的`带权路劲长度`为树中所有叶子结点的带权路径长度之和,通常记作`WPL`;


## 树的带权路劲长度
树中所有叶子节点的带权路径之和,通常记作`wpl`;
![](https://raw.githubusercontent.com/matrix-johnjohn/kimo-docs/refs/heads/main/images/articles/server/c/data-struct/tree/100.tree.png)

## 构造哈夫曼树
通过一个序列构造成哈夫曼树;

## 优先级队列offer
````c
void offer_queue(Queue* Q,TNode T){
    QNode* p=Q->front;
    QNode* s=(QNode*)malloc(sizeof(QNode));s->next=NULL;s->data=T;
    while(p->next!=NULL&&(p->next->data.weight<T.weight))p=p->next;
    if(p->next==NULL){
        p->next=s;
        Q->rear=s;
        return;
    };
    s->next=p->next;
    p->next=s;
};
````

## 优先级队列poll
````c
TNode poll_queue(Queue* Q) {
    if (Q->front->next == NULL) {
        fprintf(stderr, "Error: Poll from empty queue!\n");
        exit(EXIT_FAILURE);
    }

    QNode* s = Q->front->next;
    TNode data = s->data;

    Q->front->next = s->next;
    if (s == Q->rear) {
        Q->rear = Q->front;
    }

    free(s);
    return data;
};

````

## 构建huffman
````c
TNode* generate_huffman(Queue* Q){
    while(Q->front->next!=Q->rear){
        TNode l=poll_queue(Q);
        TNode r=poll_queue(Q);
        TNode* left=(TNode*)malloc(sizeof(TNode));*left=l;
        TNode* right=(TNode*)malloc(sizeof(TNode));*right=r;

        TNode* parent=(TNode*)malloc(sizeof(TNode));
        parent->weight=left->weight+right->weight;
        parent->left=left;
        parent->right=right;
        offer_queue(Q,*parent);
    };
    TNode res=poll_queue(Q);
    TNode* p=(TNode*)malloc(sizeof(TNode));*p=res;
    return p;
};
````

## 哈夫曼编码表
````c
void encode_helper(TNode* T, char* code, int depth) {
    if (T == NULL) return;
    if (T->left == NULL && T->right == NULL) {
        code[depth] = '\0';
        printf("%c %s\n", T->data, code);
        return;
    }
    
    code[depth] = '0';
    encode_helper(T->left, code, depth + 1);

    code[depth] = '1';
    encode_helper(T->right, code, depth + 1);
}


void encode(TNode* T) {
    if (T == NULL) return;
    char code[3] = {'\0'};
    encode_helper(T, code, 0);
}
````
