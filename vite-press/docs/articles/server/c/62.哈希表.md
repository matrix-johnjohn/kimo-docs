散列表的基本思想是查找元素记录的存储位置与关键字之间存在直接的对应关系;

## 哈希表结构
哈希表的存储结构;
```` c
typedef struct HashTable{
    int *keys;
    int length;
}HashTable,*HashTablePtr;
````

## 初始化哈希表
```` c
void InitHashTable(HashTablePtr* H){
    *H=(HashTable*)malloc(sizeof(HashTable));
    (*H)->keys=(int*)malloc(sizeof(int)*hash_val);
    
    for(int i=0;i<hash_val;i++){
        (*H)->keys[i]=empty;
    };
    (*H)->length=0;  
};
````

## 线性探测法
```` c
int hash(int key){
    return key%hash_val;
};

void put_l(HashTable* H,int key){
    if(H->length==hash_val)return;

    int index=hash(key);
    if(H->keys[index]!=empty){
        int count=1;
        while(H->keys[index]!=empty){
            index=hash(hash(key)+count);
            count++;
        };
    };
    H->keys[index]=key;
    H->length++;
};
````

## 平方探测法
```` c
void put_s(HashTable* H, int key){
    if (H->length == hash_val) 
        return;

    int start = hash(key);

    for (int i = 0; i < hash_val; i++){
        int index = (start + i * i) % hash_val;
        if (H->keys[index] == empty) {
            H->keys[index] = key;
            H->length++;
            return;
        }
        index = ((start - i * i)+hash_val) % hash_val;
        if (H->keys[index] == empty) {
            H->keys[index] = key;
            H->length++;
            return;
        }
    };
}
````
