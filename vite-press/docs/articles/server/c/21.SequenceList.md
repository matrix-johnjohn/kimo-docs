## 顺序表的概念
顺序表是用一组地址连续的存储单元依次存储线性表中的数据元素,从而使逻辑上相邻的两个元素在物理位置上也相邻;<br>
![](https://raw.githubusercontent.com/matrix-johnjohn/kimo-docs/refs/heads/main/images/articles/server/c/data-struct/linearlist/4.linearlist.png)


## 顺序表结构定义
### 数组静态分配
编译的时候分配内存,大小固定;
````C
#define MAXSIZE 100

typedef struct SQList {
	ElemType data[MAXSIZE];
	int length;
} SQList;
````

### 数组动态分配
运行的时候手动分配内存,大小可变,但是动态分配增加数组长度的时候,将原数据复制到新的空间的时候,时间和空间的开销都很大;
````C
#define MAXSIZE 100

typedef struct SQList {
	ElemType *data;
	int length;
} SQList;
````

## 顺序表初始化
初始化顺序表的函数,返回当前初始顺序表的地址;
````C
#define MAXSIZE 100

SQList* initSequenceList(){

	ElemType* element = (ElemType*)malloc(sizeof(ElemType) * MAXSIZE);
	if (element==NULL){
		printf("InitSequenceList Function Memory allocation failed\n");
		exit(EXIT_FAILURE);
	}

	SQList *sqList=(SQList*)malloc(sizeof(SQList));
	if (sqList==NULL){
		printf("InitSequenceList Function Memory allocation failed\n");
		exit(EXIT_FAILURE);
	}

	sqList->data = element;
	sqList->length = 0;

	return sqList;
}
````

## 顺序表销毁
将顺序表所占用的所有内存全部释放掉;
````C
void freeSequenceList(SQList* list){
	if (list != NULL){
		if (list->data != NULL){
			free(list->data);
		}
		free(list);
	}
}
````
## 顺序表清空
清空当前顺序表,使其变成空表;
````C 
void clearSequenceList(SQList* list){
	if (list != NULL){
		list->length = 0;
	}
}
````

## 顺序表长度
获取当前顺序表表长;
````C
int getSequenceListLength(SQList* list){
	if (list != NULL){
		return list->length;
	}
	return -1;
}
````

## 顺序表取值
根据索引获取顺序表中的数据元素;
````C
ElemType getSequenceElement(SQList *list,int index){
	if(list==NULL){
		printf("GetSequenceElement Function List is NULL\n");
		exit(EXIT_FAILURE);
	}
	if(index<0||index>=list->length){
		printf("GetSequenceElement Function Index out of bounds\n");
		exit(EXIT_FAILURE);
	}
	return list->data[index];
}
````

## 顺序表查找
顺序表线性查找,从头遍历到尾;
````C
int SequenceLinearSearch(SQList*list,ElemType e){

	for(int i=0;i<list->length;i++){
		if (list->data[i].id==e.id&&!strcmp(e.data,list->data[i].data)){
			return i;
		}
	}

	return -1;
}
````

## 顺序表插入
在顺序表尾部插入数据元素;
````C
void PushSequenceElement(SQList* list, ElemType e) {
	if (list == NULL) {//判断内存是否分配成功
		printf("PushSequenceElement Function List is NULL\n");
		exit(EXIT_FAILURE);
	}
	if (list->length >= MAXSIZE){//判断顺序表是否已满
		printf("PushSequenceElement Function List is full\n");
		exit(EXIT_FAILURE);
	}
	list->data[list->length].id = e.id;
	list->data[list->length].data = e.data;
	list->length++;
}
````

## 顺序表插入
在顺序表中的指定指定索引下插入数据元素;
````C
void InsertSequenceElement(SQList* list, int index, ElemType e){
	if(list == NULL){//判断内存是否分配成功
		printf("InsertSequenceElement Function List is NULL\n");
		exit(EXIT_FAILURE);
	}
	if (list->length >= MAXSIZE) {//判断顺序表是否已满
		printf("InsertSequenceElement Function List is full\n");
		exit(EXIT_FAILURE);
	}
	if(index<0||index >=list->length) {//判断插入位置是否合法
		printf("InsertSequenceElement Function Index out of bounds\n");
		exit(EXIT_FAILURE);
	}
	for (int i = list->length; i > index; i--) {
		list->data[i] = list->data[i - 1];
	}
	list->data[index].id = e.id;
	list->length++;
}
````

## 删除顺序表
根据索引删除顺序表中的元素;
````C
void DeleteSequenceElement(SQList* list,int index){
	if(list == NULL) {//判断内存是否分配成功
		printf("DeleteSequenceElement Function List is NULL\n");
		exit(EXIT_FAILURE);
	}
	if(index < 0 || index >= list->length) {//判断删除位置是否合法
		printf("DeleteSequenceElement Function Index out of bounds\n");
		exit(EXIT_FAILURE);
	}
	for(int i=index;i<list->length;i++){
		list->data[i] = list->data[i+1];
	}
	list->length--;
}
````
