## 分支语句
程序的执行流程是从上到下逐行执行,当你需要根据不同的条件来执行不同的事件,就需要使用分支语句;

### if
```` C
#include<stdio.h>
#define MAX 10

int main(){
	if(MAX==100)
	{
		printf("Hello, World!\n");
		//branch1
	}
	else if(MAX==10)
	{
		printf("Hello, C Programming!\n");
		//branch2
	}
	return 0;
}
````

### switch
````c
#include<stdio.h>
#define MAX 10

int main(){
	char chars[MAX]="400";
	switch (chars[0]) {
		case '1':
			printf("one\n");
			break;
		case '2':
			printf("two\n");
			break;
		default:
			printf("other\n");
			break;
	}
	return 0;
}
````

## 循环语句
当前事件需要重复执行的时候,就可以使用循环语句;

### while
````c
#include<stdio.h>
#define MAX 10

int main(){
	int num = MAX;
	while(num)
	{
		printf("current:%d\n", num);
		num--;
	}
	return 0;
}
````
如果需要至少执行一次的循环;
````c
#include<stdio.h>
#define MAX 0

int main(){
	int num = MAX;

	do {
		num++;
	} while (num);

	return 0;
}
````
### for
````c
#include<stdio.h>
#define MAX 10

int main(){
	int cur = 0;
	for(int i=0;i<MAX;i++){
		cur=i;
		printf("Index: %d\n", cur);
	}
	return 0;
}
````
