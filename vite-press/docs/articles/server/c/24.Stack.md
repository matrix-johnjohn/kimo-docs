`栈`是限定仅在`栈顶插入`或`删除`操作的线性表;栈的插入操作被称为`进栈`(push),栈的删除操作被称为`出栈`(pop);
![](https://raw.githubusercontent.com/matrix-johnjohn/kimo-docs/refs/heads/main/images/articles/server/c/data-struct/stack/01.stack.png)

## 顺序栈
顺序栈是指利用`顺序存储结构`结构实现的栈,即利用`一组地址连续的存储单元`依次存放自栈底到栈顶的数据元素;

## 顺序栈存储结构
````C
typedef struct SqStack{
    int *base;
    int *top;
    int stacksize;
}SqStack;
````

## 初始化顺序栈
````C
Status InitSqStack(SqStack **S){
    *S=(SqStack*)malloc(sizeof(SqStack));
    if((*S)==NULL)return ERROR;
    (*S)->top=(*S)->base=(int*)malloc(sizeof(int)*MAXSIZE);
    (*S)->stacksize=MAXSIZE;
    return OK;
};
````

## 顺序栈入栈
````C
Status Push(SqStack *S,int e){
    if(S->top-S->base==MAXSIZE)return ERROR;
    *S->top++=e;
    return OK;
};
````

## 顺序栈入栈
````C
Status Pop(SqStack *S,int *e){
    if(S->top==S->base)return ERROR;
    *e=*--S->top;
    return OK;
};
````

## 顺序栈取栈顶元素
````C
int GetTop(SqStack *S){
    if(S->top==S->base)return OVERFLOW;
    return *S->top-1;
};
````

## 递归遍历顺序栈
````C
void Traversal(int *top,int *base){
    if(top==base){
        printf("%d\n",*top);
        return;
    }
    printf("%d ",*top);
    Traversal(top-1,base);
};
````

## 顺序栈完整代码
````C
#define MAXSIZE 5
#include<stdio.h>
#include<stdlib.h>
#include "index.h"

typedef struct SqStack{
    int *base;
    int *top;
    int stacksize;
}SqStack;

Status InitSqStack(SqStack **S){
    *S=(SqStack*)malloc(sizeof(SqStack));
    if((*S)==NULL)return ERROR;
    (*S)->top=(*S)->base=(int*)malloc(sizeof(int)*MAXSIZE);
    (*S)->stacksize=MAXSIZE;
    return OK;
};

Status Push(SqStack *S,int e){
    if(S->top-S->base==MAXSIZE)return ERROR;
    *S->top++=e;
    return OK;
};

Status Pop(SqStack *S,int *e){
    if(S->top==S->base)return ERROR;
    *e=*--S->top;
    return OK;
};

int GetTop(SqStack *S){
    if(S->top==S->base)return OVERFLOW;
    return *S->top-1;
}

void Traversal(int *top,int *base){
    if(top==base){
        printf("%d\n",*top);
        return;
    }
    printf("%d ",*top);
    Traversal(top-1,base);
};

Status DestroySqStack(SqStack **S){
    if((*S)==NULL)return ERROR;
    free((*S)->base);
    free((*S));
    *S=NULL;
    return OK;
};

int main(){
    SqStack *S=NULL;
    InitSqStack(&S);

    for(int i=1;i<=MAXSIZE;i++){
        Push(S,i);
    };

    int *e=(int*)malloc(sizeof(int));
    Pop(S,e);
    Traversal(S->top-1,S->base);
    printf("%d %d",*e,GetTop(S));

    DestroySqStack(&S);
    return 0;
}
````

## 链栈
链栈是指采用`链式存储结构`实现的栈,通常用单链表来表示;
## 链栈存储结构
````C
typedef struct StackNode{
    int data;
    struct StackNode *next;
}StackNode,*LinkStack;
````

## 链栈初始化
````C
Status InitLinkStack(LinkStack *S){
    (*S)=NULL;
    return OK;
};
````

## 链栈入栈
````C
Status Push(LinkStack *S,int data){
    StackNode *p=(StackNode*)malloc(sizeof(StackNode));
    if(p==NULL)return ERROR;
    p->data=data;
    p->next=(*S);
    (*S)=p;
    return OK;
};
````

## 链栈出栈
````C
Status Pop(LinkStack *S,int *data){
    if(*S==NULL)return OVERFLOW;
    StackNode *p=*S;
    *S=p->next;
    *data=p->data;
    free(p);
    return OK;
};
````

## 获取链栈头元素
````C
int GetTop(LinkStack S){
    if(S==NULL)return OVERFLOW;
    return S->data;
}
````

## 递归遍历链栈
````C
void Traversal_R(LinkStack S){
    if(S==NULL){
        printf("\n");
        return;
    };
    printf("%d ",S->data);
    Traversal_R(S->next);
};
````

## 链栈完整代码实现
````C
#include<stdio.h>
#include<stdlib.h>
#include "common.h"

typedef struct StackNode{
    int data;
    struct StackNode *next;
}StackNode,*LinkStack;

Status InitLinkStack(LinkStack *S){
    (*S)=NULL;
    return OK;
};

Status Push(LinkStack *S,int data){
    StackNode *p=(StackNode*)malloc(sizeof(StackNode));
    if(p==NULL)return ERROR;
    p->data=data;
    p->next=(*S);
    (*S)=p;
    return OK;
};

Status Pop(LinkStack *S,int *data){
    if(*S==NULL)return OVERFLOW;
    StackNode *p=*S;
    *S=p->next;
    *data=p->data;
    free(p);
    return OK;
};

int GetTop(LinkStack S){
    if(S==NULL)return OVERFLOW;
    return S->data;
};

void Traversal_R(LinkStack S){
    if(S==NULL){
        printf("\n");
        return;
    };
    printf("%d ",S->data);
    Traversal_R(S->next);
};

int main(){
    LinkStack S=NULL;
    int *data=(int*)malloc(sizeof(int));
    InitLinkStack(&S);
    for(int i=1;i<=5;i++){
        Push(&S,i);
    };
    Pop(&S,data);
    Traversal_R(S);
    printf("%d %d\n",*data,GetTop(S));
    Traversal_R(S);
    return 0;
}
````
