## Stack
`栈`是限定仅在`栈顶插入`或`删除`操作的线性表;栈的插入操作被称为`进栈`(push),栈的删除操作被称为`出栈`(pop);
![](https://raw.githubusercontent.com/matrix-johnjohn/kimo-docs/refs/heads/main/images/articles/server/c/data-struct/stack/01.stack.png)

## 顺序栈
顺序栈是指利用`顺序存储结构`结构实现的栈,即利用`一组地址连续的存储单元`依次存放自栈底到栈顶的数据元素;

## 顺序栈的结构
````C
typedef struct SqStack{
	int* base;//栈底指针
	int* top;//栈顶指针
	int stacksize;//栈可用的最大容量
}SqStack;
````

## 初始化顺序栈
````C
Status InitStack(SqStack *S){
	S->base=(int*)malloc(sizeof(int)*MAXSIZE);
	if (!S->base){
		return OVERFLOW;
	};
	S->top=S->base;
	S->stacksize=MAXSIZE;
	return OK;
};
````

## 顺序栈入栈
````C
void Push(SqStack* S, int data){
	if(S->top-S->base==S->stacksize){
		exit(OVERFLOW);
	}
	*S->top++=data;
};
````

## 顺序栈入栈
````C
void Pop(SqStack *S,int *num){
	if (S->top == S->base) {
		exit(OVERFLOW);
	};
	*num=*--(S->top);
};
````

## 顺序栈取栈顶元素
````C
int GetTop(SqStack *S){
	if (S->top == S->base) {
		exit(OVERFLOW);
	};
	return *(S->top-1);
};
````

## 链栈
链栈是指采用`链式存储结构`实现的栈,通常用单链表来表示;
````C
typedef struct StackNode {
	int data;
	struct StackNode *next;
}StackNode,*LinkStack;
````

## 链栈初始化
````C
Status InitStack(LinkStack *S){
	S = NULL;
	return OK;
};
````

## 链栈入栈
````C
Status Push(LinkStack *S,int e){
	StackNode *p=(StackNode*)malloc(sizeof(StackNode));
	p->data = e;
	p->next=(*S);
	*S = p;
	return OK;
};
````

## 链栈出栈
````C
Status Pop(LinkStack *S,int *e){
	if(*S==NULL) return ERROR;
	*e=(*S)->data;
	StackNode *p=*S;
	(*S)=(*S)->next;
	free(p);
};
````

## 链栈完整代码实现
````C
#include<stdio.h>
#include<stdlib.h>
#define	MAXSIZE 100
#include "index.h"

typedef struct StackNode{
	int data;
	struct StackNode *next;
}StackNode,*LinkStack;

Status InitStack(LinkStack *S){
	S = NULL;
	return OK;
};

Status Push(LinkStack *S,int e){
	StackNode *p=(StackNode*)malloc(sizeof(StackNode));
	p->data = e;
	p->next=(*S);
	*S = p;
	return OK;
};

Status Pop(LinkStack *S,int *e){
	if(*S==NULL) return ERROR;
	*e=(*S)->data;
	StackNode *p=*S;
	(*S)=(*S)->next;
	free(p);
};

void Traversal(LinkStack *S){
	StackNode *p=*S;
	while (p){
		printf("%d ",p->data);
		p = p->next;
	}
	printf("\n");
}

int main() {
	int* e=(int*)malloc(sizeof(int));
	LinkStack S=NULL;
	InitStack(&S);
	for (int i = 1; i <= 10; i++) {
		Push(&S,i);
	};
	Pop(&S, e);
	Traversal(&S);
	printf("Popped element: %d\n", *e);
	return 0;
};
````
