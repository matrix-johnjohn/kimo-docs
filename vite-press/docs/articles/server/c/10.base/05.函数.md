函数是程序组织的基本单元,是实现模块化编程的核心机制;通过预定义的接口接收输入,执行特定的计算或操作,并通过控制流返回结果或状态;<br>

函数调用的时候,会在栈内存中分配一块空间内存,用来存储函数内部的局部变量或者函数的参数,然后在函数调用结束后,这块内存空间会被释放掉;

## 返回
````C
#include<stdio.h>
#include<stdlib.h>

void traversal(int *array,int size) {//no return
	for (int i = 0; i < size; i++) {
		printf("%d ", array[i]);
	}
}

int sum(int* array,int size){//return
	int result = 0;
	for (int i = 0; i < size; i++) {
		result+=array[i];
	}
	return result;
}

int main(){
	int len = 5;
	int* array = (int*)malloc(len * sizeof(int));
	if (array == NULL) {
		fprintf(stderr, "内存分配失败。\n");
		return 1;
	}
	for (int i = 0; i < 5; i++) {
		array[i] = i + 1;
	}

	traversal(array,len);

	int result=sum(array, len);

	free(array);
	array = NULL;
	return 0;
}

````

## 参数
C的函数的传参方式有两种,一种是`传值`,一种是`传地址`;如果传输的是值,只是把`实参`的值浅拷贝给`形参`,这种情况下,形参的改变并不会直接影响到实参;
````C
#include<stdio.h>
#include<stdlib.h>
#include "main.h"

int main() {
	int num1 = 10;
	int num2 = 20;
	swap(&num1, &num2);

	printf("%d %d\n", num1, num2);
	return 0;
}

void swap(int* a, int* b) {
	int temp = *a;
	*a = *b;
	*b = temp;
}
````

## static
static关键字用于修饰函数内部局部变量,表示该变量为一个静态变量,使其生命和程序的生命周期一致,并不会随着函数的调用结束而被释放,第二次调用该函数时,变量的值依旧保留;
````C
#include<stdio.h>
#include<stdlib.h>
#include "main.h"

void global() {
	static int count = 0;
	count++;
	printf("%d\n", count);
}

int main() {
	global();
	global();
	global();
	
	return 0;
}
````

## 函数指针
函数指针是 C 语言中一种特殊的指针类型，它指向一个函数，而不是普通变量。通过函数指针，可以在运行时动态调用不同的函数，常用于回调、事件处理、状态机、库接口等场景；
````C
#define _CRT_SECURE_NO_WARNINGS
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
int add(int a, int b);
int add(int a, int b) {
    return a + b;
};

int main() {
    int (*p)(int, int);//声明函数指针
    p = &add;//地址传输

    int s=p(1, 3);
    printf("%d",s);

    return 0;
}
````
