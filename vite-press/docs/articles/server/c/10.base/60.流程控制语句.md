## 分支语句
程序的执行流程是从上到下逐行执行,当你需要根据不同的条件来执行不同的事件,就需要使用分支语句;

### if
```` C
#include<stdio.h>
#define MAX 10

int main(){
	if(MAX==100)
	{
		printf("branch1!\n");
		//branch1
	}
	else if(MAX==10)
	{
		printf("branch2!\n");
		//branch2
	}
	return 0;
}
````

### switch
````c
#include<stdio.h>
#define MAX 10

int main(){
	char chars[MAX]="400";
	switch (chars[0]) {
		case '1':
			printf("one\n");
			break;
		case '2':
			printf("two\n");
			break;
		default:
			printf("other\n");
			break;
	}
	return 0;
}
````

## 循环语句
当前事件需要重复执行的时候,就可以使用循环语句;

### while
````c
#include<stdio.h>
#define MAX 10

int main(){
	int num = MAX;
	while(num)
	{
		printf("current:%d\n", num);
		num--;
	}
	return 0;
}
````
如果需要至少执行一次的循环;
````c
#include<stdio.h>
#define MAX 0

int main(){
	int num = MAX;

	do {
		num++;
	} while (num);

	return 0;
}
````
### for
````c
#include<stdio.h>
#define MAX 10

int main(){
	int cur = 0;
	for(int i=0;i<MAX;i++){
		cur=i;
		printf("Index: %d\n", cur);
	}
	return 0;
}
````

## 数组
数组是一种基础的顺序存储结构的数据结构,可以通过下标进行访问或修改;
```` C
#include<stdio.h>
#include<stdlib.h>

int main(){
	int array[5] = {1,2,3,4,5};
	for (int i = 0; i < 5; i++)
	{
		printf("%d\n", array[i]);
	}
	return 0;
}
````
````C
#include<stdio.h>
#include<stdlib.h>
#define MAX_SIZE 10

int main() {
	int* array = (int*)malloc(sizeof(int) * MAX_SIZE);
	for (int i = 0; i < MAX_SIZE; i++)
	{
		*array = i+1;
		array=array+1;
	}
	array -= MAX_SIZE;
	for (int i = 0; i < MAX_SIZE; i++)
	{
		printf("%d ", array[i]);
	}
	free(array);
	array = NULL;
	return 0;
}
````
