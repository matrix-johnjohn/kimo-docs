## 继承
`面向对象程序设计`中最重要的一个概念是`继承`。继承允许我们依据另一个类来定义一个类，这使得创建和维护一个应用程序变得更容易。这样做，也达到了重用代码功能和提高执行效率的效果。

当创建一个类时，您不需要重新编写新的数据成员和成员函数，只需指定新建的类继承了一个已有的类的成员即可。这个已有的类称为基类，新建的类称为派生类。
````cpp
#include<iostream>

using namespace std;

class Sharp{//基础类
    protected:
        int width;
        int height;
    public:
        void setWidth(int width){
            this->width=width;
        };
        void setHeight(int height){
            this->height=height;
        };
};

class Rectangle:public Sharp{//派生类
    public:
        void setWidth(int width){
            this->width=width;
            cout<<"set width"<<endl;
        };
        void setHeight(int height){
            this->height=height;
            cout<<"set height"<<endl;
        };
        int getArea(){
            return this->width*this->height;
        }
};

int main(){
    Rectangle *p1=new Rectangle();
    p1->setHeight(15);
    p1->setWidth(10);
    int area=p1->getArea();
    cout<<area<<endl;
    return 0;
};
````

## 虚函数
````cpp
#include<iostream>

using namespace std;

class Entity{
    public:
        virtual string getName(){//虚函数
            return "entity";
        };
};

class Player:public Entity{
    private:
        string name;
    public:
        Player(const string &n):name(n){};

        string getName()override{
            return this->name;
        }
};

int main(){
    Entity *e=new Entity;
    cout<<e->getName()<<endl;

    Player *p=new Player("Hello");
    cout<<p->getName()<<endl;

    Entity *en=new Entity;
    en=p;
    cout<<en->getName()<<endl;
    return 0;
};
````

## 纯虚函数
````cpp
#include<iostream>

using namespace std;

class Entity{
protected:
    int *data;
public:
    Entity(){};
    Entity(int data){
        this->data=new int(data);
    };
    virtual void setData(int data)=0;//纯虚函数
    virtual ~Entity()=default;//虚析构函数
};

class Table:public Entity{
    public:
        Table(int d);
        void setData(int data)override{
            *(this->data)=data;
        };
        int getData(){
            return *(this->data);
        };
        ~Table()override{
            delete this->data;
        }
};

Table::Table(int d):Entity(d){};//显式调用基类构造函数

int main(){
    Table *t=new Table(20);
    t->setData(25);
    int data=t->getData();
    cout<<data<<endl;
    return 0;
};
````
