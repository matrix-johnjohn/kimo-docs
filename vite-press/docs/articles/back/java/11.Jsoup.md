## 爬虫实现
获取网站图标的api
````java
@Test
void test2() {
    // 目标网站
    String websiteUrl = "https://docs.expo.dev/"; // 网站 url

    // Todo: 通过 url 获取图标的 URL
    String faviconUrl = getFaviconUrl(websiteUrl);

        // 输出结果
    System.out.println("Website: " + websiteUrl);
    System.out.println("Favicon URL: " + faviconUrl);

        // 断言：确保获取到了有效的 URL
    assertTrue(faviconUrl != null && !faviconUrl.isEmpty(), "Favicon URL should not be null or empty");
    assertTrue(faviconUrl.startsWith("http"), "Favicon URL should be absolute (starts with http)");
    }

    /**
     * 从指定网站 HTML 中解析出 favicon 的 URL
     *
     * @param websiteUrl 网站主页地址
     * @return favicon 的完整绝对 URL，如果没找到则返回 null
     */
private String getFaviconUrl(String websiteUrl) {
    try {
            // 1. 下载网页 HTML
            Document doc = Jsoup.connect(websiteUrl).get();
            // 2. 查找所有可能的 icon 标签
            Elements icons = doc.select("link[rel~=(?i)^(shortcut )?icon]");
            String faviconPath = null;
            for (Element link : icons) {
                faviconPath = link.attr("href");
                if (faviconPath != null && !faviconPath.isEmpty()) {
                    break;
                }
            }
            // 3. 如果没找到，则尝试默认路径
            if (faviconPath == null || faviconPath.isEmpty()) {
                faviconPath = "/favicon.ico";
            }

            // 4. 将相对路径转为绝对 URL
            URL baseUrl = new URL(websiteUrl); // 基础 URL
            URL resolvedUrl = new URL(baseUrl, faviconPath); // 自动处理相对路径

            // 5. 转成字符串并处理编码（如空格 -> %20）
            return URI.create(resolvedUrl.toString()).toASCIIString();

        } catch (Exception e) {
            System.err.println("解析 favicon URL 失败: " + e.getMessage());
            return null;
        }
    }
````
